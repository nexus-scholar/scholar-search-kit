\begin{frame}{Parsing Authors \& Dates}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Author Splitting}
        \begin{itemize}
          \item Unpacks string names into \texttt{given\_name} and \texttt{family\_name}.
          \item Preserves single-name scholars (e.g. "Plato", "Aristotle").
          \item Captures ORCID identifiers.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Date Normalization}
        \begin{itemize}
          \item Parses ISO-8601, RFC-3339, and Crossref \texttt{date-parts}.
          \item Handles PubMed XML \texttt{<PubDate>}.
          \item Strictly defaults to \texttt{None} if year is unknown.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{\texttt{Author} Model}
\begin{lstlisting}[style=nexuspython]
@dataclass
class Author:
    family_name: str
    given_name: str | None = None
    orcid: str | None = None

    @property
    def full_name(self) -> str:
        return f"{self.given_name} {self.family_name}" if self.given_name else self.family_name
\end{lstlisting}

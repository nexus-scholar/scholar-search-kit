class SearchProvider(Protocol):

class BaseProvider(ABC):
    def __init__(self, name: str, rate_limit: float = 5.0):
        self.name = name
        self.http = AcademicHttpClient(name=name, rate_limit=rate_limit)

    @abstractmethod
    def search(self, query: Query) -> Iterator[Document]: ...

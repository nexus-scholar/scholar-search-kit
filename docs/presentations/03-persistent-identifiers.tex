\begin{frame}[fragile]{The \texttt{ExternalIds} Contract}
\begin{lstlisting}[style=nexuspython]
@dataclass
class ExternalIds:
    doi: str | None = None
    arxiv_id: str | None = None
    pubmed_id: str | None = None
    openalex_id: str | None = None
    s2_id: str | None = None

    def __post_init__(self) -> None:
        if self.doi:
            val = self.doi.strip().lower()
            for p in ("https://doi.org/", "http://doi.org/", "doi:"):
                if val.startswith(p): val = val[len(p):]
            self.doi = val.strip() or None
\end{lstlisting}
\end{frame}

\begin{frame}{Invariants \& Guarantees}
  \begin{itemize}
    \item \textbf{Case-Insensitive}: Lowercases all DOI strings deterministically.
    \item \textbf{Prefix Stripping}: URL schemes and URN prefixes are stripped.
    \item \textbf{Whitespace Cleansing}: Blank strings become \texttt{None}.
    \item \textbf{No Cross-Contamination}: Separate fields for DOI, arXiv, PubMed, OpenAlex, and S2 Corpus IDs.
  \end{itemize}
\end{frame}

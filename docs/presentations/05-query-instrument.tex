\begin{frame}{Query as a Formal Instrument}
  \begin{alertblock}{Scientific Auditability}
    In reproducible research, search parameters cannot be transient command line arguments. They must be immutable, structured research objects.
  \end{alertblock}
  \vspace{0.3cm}
  \begin{center}
  \resizebox{0.85\linewidth}{!}{
  \begin{tikzpicture}[node distance=1cm and 1.5cm]
    \node[primarycard] (q) {\textbf{\texttt{Query}}\\ID: \texttt{Q001}\\Text: \texttt{"deep learning" AND vision}\\Years: 2020--2024\\Max: 100};
    
    \node[successcard, right=of q] (doc) {\textbf{Retrieved \texttt{Document}}\\Title: "Vision Transformers..."\\\texttt{query\_id: "Q001"}\\\texttt{retrieved\_at: 2026-08-21T...}};

    \draw[accentarrow] (q) -- node[above, font=\scriptsize\bfseries] {Provenance Link} (doc);
  \end{tikzpicture}
  }
  \end{center}
\end{frame}

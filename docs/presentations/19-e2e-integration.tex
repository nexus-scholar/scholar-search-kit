\begin{frame}[fragile]{Complete Python Automation Script}
\begin{lstlisting}[style=nexuspython]
from scholar_search import SearchEngine, DocumentVerifier, Exporter, Query

# 1. Search across global repositories
engine = SearchEngine(providers=["openalex", "crossref", "arxiv"])
results = engine.search(Query("multimodal foundation models", max_results=30))

# 2. Verify against Crossref & Hydrate
verifier = DocumentVerifier()
verified = [d for d in results if verifier.verify_document(d).is_verified]

# 3. Export to JSON for scholar-pdf-kit
Exporter().json(verified, "verified_literature.json")
\end{lstlisting}

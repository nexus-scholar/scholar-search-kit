\begin{frame}[fragile]{Crossref Provider Implementation}
\begin{lstlisting}[style=nexuspython]
class CrossrefProvider(BaseProvider):
    def validate_reference(self, citation: str) -> dict | None:
        url = "https://api.crossref.org/works"
        params = {"query.bibliographic": citation, "rows": 1}
        resp = self.http.get(url, params=params).json()
        items = resp.get("message", {}).get("items", [])
        return items[0] if items else None
\end{lstlisting}

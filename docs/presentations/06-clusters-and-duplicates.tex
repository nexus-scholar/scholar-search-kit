\begin{frame}{Non-Destructive Deduplication}
  \begin{alertblock}{Preventing Evidence Destruction}
    Naive deduplication permanently deletes duplicate records. We preserve every merged record inside a \texttt{DocumentCluster} while synthesizing an enriched representative.
  \end{alertblock}

\begin{frame}[fragile]{The \texttt{DocumentCluster} Dataclass}
\begin{lstlisting}[style=nexuspython]
@dataclass
class DocumentCluster:
    cluster_id: int
    representative: Document
    members: list[Document]

    @property
    def size(self) -> int:
        return len(self.members)

    @property
    def confidence(self) -> float:
        has_id = any(
            m.external_ids.doi or m.external_ids.arxiv_id 
            for m in self.members
        )
        return 1.0 if has_id else 0.95
\end{lstlisting}

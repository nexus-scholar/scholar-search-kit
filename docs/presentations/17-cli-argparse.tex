\begin{frame}[fragile]{Interactive \& Batch CLI Usage}
\begin{lstlisting}[style=nexuspython]
# 1. Federated Search across OpenAlex, PubMed, and Crossref
scholar-search search "attention mechanisms" --limit 20 --output results.json

# 2. Forward & Backward Citation Snowballing
scholar-search snowball W2741809807 --direction forward --limit 50

# 3. Verified Legacy Ingestion
scholar-search import citations.ris --verify --enrich --output clean.json
\end{lstlisting}

\begin{frame}{The Specification Contract}
  \begin{alertblock}{Design by Contract}
    Scientific software must not have ambiguous behaviors. We define explicit contracts: inputs, outputs, mathematical invariants, and failure modes.
  \end{alertblock}

  \begin{center}

\begin{frame}{Key Contract Invariants}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Identifier Invariants}
        \begin{itemize}
          \item Case-insensitive DOI matching.
          \item Prefix stripping (\texttt{doi:}, URLs).
          \item Nullable empty strings.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Temporal Invariants}
        \begin{itemize}
          \item Explicit UTC timestamps.
          \item Year bounded ($1900 \le y \le 2100$).
          \item Non-zero publication dates.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
